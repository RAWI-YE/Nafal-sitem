تنبيه صلاحيات قاعدة البيانات (Firebase Rules)

انسخ هذا الكود وضعه في موقع Firebase Console داخل قسم Firestore Database ثم تبويب Rules واضغط Publish: 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
